Zephyr SDK 0.17.4 Setup
0. -> wget https://github.com/zephyrproject-rtos/sdk-ng/releases/download/v0.17.4/zephyr-sdk-0.17.4_linux-x86_64.tar.xz
tar xvf zephyr-sdk-0.17.4_linux-x86_64.tar.xz
cd zephyr-sdk-0.17.4/
./setup.sh

1. mkdir ZRTOS_demo1 
~/wsMA_prototyp$ python3 -m venv venv_zrtos_demo1
source '[...]/wsMA_prototyp/venv_zrtos_demo1/bin/activate'
sudo apt update
sudo apt upgrade
sudo apt install --no-install-recommends ninja-build gperf ccache dfu-util device-tree-compiler xz-utils file make gcc gcc-multilib g++-multilib libsdl2-dev libmagic1
pip install pyelftools (was needed later for west build native_sim ...)
pip install openpyxl
pip install west

west init -m https://github.com/zephyrproject-rtos/zephyr --mr v4.2.1
